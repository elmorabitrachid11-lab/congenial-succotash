<script>
function handleLogin() {
const email = document.getElementById('email').value;
const pass = document.getElementById('password').value;
if(!email || !pass) {
const msg = document.getElementById('feedback-msg');
msg.textContent = "Please enter both email and password";
msg.classList.remove('hidden');
return;
}
document.getElementById('auth-screen').classList.add('hidden');
document.getElementById('app-screen').classList.remove('hidden');
}
function handleRegister() {
alert("Redirecting to account registration.");
}
function handleLogout() {
document.getElementById('app-screen').classList.add('hidden');
document.getElementById('auth-screen').classList.remove('hidden');
document.getElementById('email').value = "";
document.getElementById('password').value = "";
document.getElementById('feedback-msg').classList.add('hidden');
}
function switchTab(tab) {
['feed', 'dashboard', 'profile'].forEach(t => {
const btn = document.getElementById('tab-' + t);
if(t === tab) {
btn.className = "text-white border-b-2 border-red-600 pb-1 font-bold";
} else {
btn.className = "text-zinc-400 pb-1";
}
});
}
function openLiveStream() { alert("Opening live stream..."); }
function openGiftModal() { alert("Sending support gift."); }
function reportInappropriateContent() { alert("Content reported successfully."); }
